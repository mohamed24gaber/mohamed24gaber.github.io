#define KEY_WIDTH 3.0
#define KEY_HEIGHT 3.0
#define GAP 0.5

// Final separations
#define X_SENSOR_X_POS 45.5
#define Y_SENSORS_Y_POS -10.0

// Sensor pins: {X, Y1, Y2, Y3}
const int trigPins[4] = {2, 4, 6, 8};
const int echoPins[4] = {3, 5, 7, 9};

// QWERTY key layout
char keys[NUM_ROWS][NUM_COLS] = {
  {'Q','W','E','R','T','Y','U','I','O','P'},
  {'A','S','D','F','G','H','J','K','L',';'},
  {'Z','X','C','V','B','N','M',',','.','/'},
  {'1','2','3','4','5','6','7','8','9','0'}
};

float measureDistance(int trigPin, int echoPin) {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH, 25000);

  if (duration == 0) return -1;

  return duration * 0.0343 / 2.0;
}

void setup() {
  Serial.begin(9600);

  for (int i = 0; i < 4; i++) {
    pinMode(trigPins[i], OUTPUT);
    pinMode(echoPins[i], INPUT);
  }
}

void loop() {
  // 1. Measure X (from right to left)
  float xDist = measureDistance(trigPins[0], echoPins[0]);

  if (xDist < 0 || xDist > 100) return;

  float xFromLeft = X_SENSOR_X_POS - xDist;

  if (xFromLeft < 0 || xFromLeft > 35.5) return;

  int col = xFromLeft / (KEY_WIDTH + GAP);

  if (col < 0 || col >= NUM_COLS) return;

  // 2. Measure Y from three sensors
  float yDists[3];

  for (int i = 0; i < 3; i++) {
    yDists[i] = measureDistance(
      trigPins[i + 1],
      echoPins[i + 1]
    );
  }

  // 3. Find the closest valid Y reading
  int ySensorIndex = -1;
  float minY = 1000;

  for (int i = 0; i < 3; i++) {
    if (yDists[i] > 1.0 && yDists[i] < minY) {
      minY = yDists[i];
      ySensorIndex = i;
    }
  }

  if (ySensorIndex == -1) return;

  float yFromTop = minY - (-Y_SENSORS_Y_POS);

  if (yFromTop < 0 || yFromTop > 14.5) return;

  int row = yFromTop / (KEY_HEIGHT + GAP);

  if (row < 0 || row >= NUM_ROWS) return;

  char key = keys[row][col];

  Serial.print("Key Pressed: ");
  Serial.println(key);

  delay(300); // debounce
}
