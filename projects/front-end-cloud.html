import { useEffect, useId, useRef, useState } from 'react'

/* ---------- Day theme: colours are interpolated from the hour ---------- */

const KEYS = [
  { h: 0, bg: [11, 16, 48], glow: [255, 176, 96] },
  { h: 6, bg: [255, 180, 162], glow: [255, 150, 90] },
  { h: 12, bg: [205, 232, 255], glow: [255, 255, 255] },
  { h: 18, bg: [90, 62, 150], glow: [255, 190, 110] },
  { h: 24, bg: [11, 16, 48], glow: [255, 176, 96] },
]

const mix = (a, b, t) => a.map((v, i) => v + (b[i] - v) * t)
const rgb = (c) => `rgb(${c.map(Math.round).join(',')})`

function themeAt(h) {
  let i = 0
  while (i < KEYS.length - 2 && h > KEYS[i + 1].h) i++

  const a = KEYS[i], b = KEYS[i + 1]
  const t = (h - a.h) / (b.h - a.h)

  const bg = mix(a.bg, b.bg, t)
  const glow = mix(a.glow, b.glow, t)

  const lum =
    (0.299 * bg[0] + 0.587 * bg[1] + 0.114 * bg[2]) / 255

  return {
    bg: rgb(bg),
    glow: rgb(glow),
    ink: lum > 0.5 ? '#1b1633' : '#fdf3e3',
    night: Math.min(1, Math.max(0, (0.7 - lum) / 0.55)),
  }
}

const fmt = (h) => {
  const m = Math.round((h % 24) * 60) % 1440
  const hh = Math.floor(m / 60)

  return `${((hh + 11) % 12) + 1}:${String(m % 60).padStart(2, '0')} ${hh < 12 ? 'am' : 'pm'}`
}

const moment = (h) =>
  h < 5 ? 'Deep night'
  : h < 7 ? 'Sunrise'
  : h < 11 ? 'Morning'
  : h < 14 ? 'Midday'
  : h < 17 ? 'Afternoon'
  : h < 19.5 ? 'Golden hour'
  : h < 22 ? 'Evening'
  : 'Night'

/* ---------- Hook: the hour, with an eased tween ---------- */

function useDay() {
  const [h, setH] = useState(4.5)
  const hRef = useRef(4.5)
  const raf = useRef(0)

  const set = (v) => {
    hRef.current = v
    setH(v)
  }

  const stop = () => cancelAnimationFrame(raf.current)

  const goTo = (to, ms = 1500) => {
    stop()

    const from = hRef.current
    const t0 = performance.now()

    const step = (t) => {
      const p = Math.min(1, (t - t0) / ms)
      const e =
        p < 0.5
          ? 2 * p * p
          : 1 - Math.pow(-2 * p + 2, 2) / 2

      set(from + (to - from) * e)

      if (p < 1) {
        raf.current = requestAnimationFrame(step)
      }
    }

    raf.current = requestAnimationFrame(step)
  }

  useEffect(() => {
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      set(17.5)
      return
    }

    const t = setTimeout(() => goTo(17.5, 4500), 500)

    return () => {
      clearTimeout(t)
      stop()
    }
  }, [])

  return { h, set, goTo, stop }
}

/* ---------- Lamp drawing ---------- */

function Lamp({ shape = 'orb', power }) {
  const id = useId().replace(/:/g, '')
  const baseX = shape === 'arc' ? 96 : 60

  return (
    <svg viewBox="0 0 120 170" className="lamp" aria-hidden="true">
      <defs>
        <radialGradient id={id}>
          <stop offset="0" stopColor="var(--glow)" stopOpacity="0.95" />
          <stop offset="1" stopColor="var(--glow)" stopOpacity="0" />
        </radialGradient>
      </defs>

      <circle
        cx="60"
        cy="62"
        r={30 + 34 * power}
        fill={`url(#${id})`}
        opacity={0.15 + 0.85 * power}
      />

      {shape === 'orb' &&
        <circle cx="60" cy="62" r="30" className="shade" />}

      {shape === 'cone' &&
        <path d="M32 92 L48 36 H72 L88 92 Z" className="shade" />}

      {shape === 'arc' && (
        <>
          <path
            d="M96 150 C96 60 70 36 44 50"
            className="stem"
          />

          <path
            d="M24 58 A22 22 0 0 1 68 58 Z"
            className="shade"
            transform="rotate(-14 46 58)"
          />
        </>
      )}

      {shape !== 'arc' &&
        <rect
          x="57"
          y="92"
          width="6"
          height="52"
          className="ink"
        />}

      <ellipse
        cx={baseX}
        cy="150"
        rx="26"
        ry="7"
        className="ink"
      />
    </svg>
  )
}

/* ---------- Hero ---------- */

const SCENES = [
  { n: 'Wake', h: 6.5 },
  { n: 'Focus', h: 12 },
  { n: 'Wind down', h: 20.5 },
]

function Hero({ day, th }) {
  const { h, set, goTo, stop } = day

  const pos = (hh) => {
    const a = ((hh - 6) / 24) * Math.PI * 2

    return {
      left: `${50 - 42 * Math.cos(a)}%`,
      top: `${94 - 80 * Math.sin(a)}%`,
    }
  }

  const sun = pos(h)
  const moon = pos((h + 12) % 24)

  return (
    <section className="hero">

      <div className="copy">
        <h1>Light that keeps your hours.</h1>

        <p className="lede">
          Dayline lamps shift their colour with the time of day,
          warm at dawn, bright at noon, amber at night.
          Drag the day below and watch the whole page follow.
        </p>
      </div>

      <div className="sky" style={{ '--night': th.night }}>
        <span className="stars" />
        <span className="sun" style={sun} />
        <span className="moon" style={moon} />
        <Lamp shape="orb" power={th.night} />
      </div>

      <div className="controls">

        <div className="clock" aria-live="off">
          <strong>{fmt(h)}</strong>
          <span>{moment(h)}</span>
        </div>

        <input
          className="day"
          type="range"
          min="0"
          max="24"
          step="0.05"
          value={h}
          aria-label="Time of day"
          onChange={(e) => {
            stop()
            set(+e.target.value)
          }}
        />

        <div className="scenes">
          {SCENES.map((s) => (
            <button
              key={s.n}
              aria-pressed={Math.abs(h - s.h) < 1.5}
              onClick={() => goTo(s.h)}
            >
              {s.n}
            </button>
          ))}
        </div>

      </div>

    </section>
  )
}

function Card({ lamp, power }) {
  const ref = useRef(null)

  const move = (e) => {
    const r = ref.current.getBoundingClientRect()

    ref.current.style.setProperty(
      '--ry',
      `${((e.clientX - r.left) / r.width - 0.5) * 14}deg`
    )

    ref.current.style.setProperty(
      '--rx',
      `${-((e.clientY - r.top) / r.height - 0.5) * 14}deg`
    )
  }

  const reset = () => {
    ref.current.style.setProperty('--rx', '0deg')
    ref.current.style.setProperty('--ry', '0deg')
  }

  return (
    <article
      className="card"
      ref={ref}
      onPointerMove={move}
      onPointerLeave={reset}
    >

      <Lamp shape={lamp.shape} power={power} />

      <h3>{lamp.name}</h3>
      <p>{lamp.note}</p>

      <div className="meta">
        <span>{lamp.k}</span>
        <b>{lamp.price}</b>
      </div>

    </article>
  )
}

/* ---------- Waitlist ---------- */

function Waitlist() {
  const [email, setEmail] = useState('')
  const [status, setStatus] = useState('idle')

  const submit = (e) => {
    e.preventDefault()

    setStatus(
      /^\S+@\S+\.\S+$/.test(email)
        ? 'done'
        : 'error'
    )
  }

  if (status === 'done')
    return (
      <div className="done" role="status">

        <svg viewBox="0 0 52 52" aria-hidden="true">
          <circle cx="26" cy="26" r="23" />
          <path d="M15 27l8 8 14-17" />
        </svg>

        <p>
          You are on the list. We will write when the first batch ships.
        </p>

      </div>
    )

  return (
    <form onSubmit={submit} noValidate>

      <label htmlFor="em">Email</label>

      <div className="row">

        <input
          id="em"
          type="email"
          value={email}
          placeholder="you@example.com"
          onChange={(e) => {
            setEmail(e.target.value)
            setStatus('idle')
          }}
          aria-invalid={status === 'error'}
        />

        <button type="submit">
          Join the waitlist
        </button>

      </div>

      <p className="err" role="alert">
        {status === 'error'
          ? 'Enter a valid email, like you@example.com.'
          : ''}
      </p>

    </form>
  )
}
